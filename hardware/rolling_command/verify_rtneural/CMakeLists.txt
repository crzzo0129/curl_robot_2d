cmake_minimum_required(VERSION 3.16)
project(verify_rolling_rtneural LANGUAGES CXX)
set(RTNEURAL_SOURCE "" CACHE PATH "Robot controller's modules/RTNeural directory")
if(NOT EXISTS "${RTNEURAL_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Set RTNEURAL_SOURCE to the installed controller's RTNeural sources")
endif()
set(RTNEURAL_EIGEN ON CACHE BOOL "Match neural_controller's Eigen backend" FORCE)
add_subdirectory("${RTNEURAL_SOURCE}" rtneural)
add_executable(verify_rolling_rtneural main.cpp)
target_compile_features(verify_rolling_rtneural PRIVATE cxx_std_17)
target_link_libraries(verify_rolling_rtneural PRIVATE RTNeural)
